## 6. 验收操作步骤汇总（命令速查）

```powershell
# 环境准备（开发模式）
npm run tauri dev

# ── 服务开关（L4 手测前置；默认 multi_device_enabled=false = 不监听）──
#   开启：设置 → 设备 → 「启用多端服务」（multi_device_set_enabled(true)，
#   服务以 wss-only 监听）；关闭：同一开关（停服不 bind，红线 1）
# ── wss 证书忽略参数（自签证书）：curl -k / wscat -n / websocat -k ──

# L4-1：服务自报（服务开启后；wss-only）
curl -k https://127.0.0.1:8765/v1/info

# L4-2：401（无令牌 / 坏令牌）
curl -k -i https://127.0.0.1:8765/v1/status
curl -k -i -H "Authorization: Bearer deadbeef" https://127.0.0.1:8765/v1/status

# L4-3：配对拿令牌（POST /v1/pair 或 WS pair.hello；令牌 64 hex，仅此一次明文）
curl -k -X POST https://127.0.0.1:8765/v1/pair -H "Content-Type: application/json" -d '{"code":"CXN-XXX"}'
#   → 200 {"token":"<64hex>","label":...}；错码 401 / 连错 5 次 409（锁定）
#   wscat -n -c "wss://127.0.0.1:8765/v1/events"   # 发 pair.hello 帧 → pair.ok + device.ready

# L4-4：事件流订阅（$TOKEN = 配对令牌）
#   wscat -n -c "wss://127.0.0.1:8765/v1/events?token=$TOKEN"
#   发 {"v":1,"kind":"events.subscribe","msgId":"t1","body":{"sessions":["<sessionId>"]}}
#   桌面发起 run → 断言 events.push

# L4-5：手机 approve（$TID = 待审批 tool_call_id）
curl -k -X POST https://127.0.0.1:8765/v1/approval/$TID/approve -H "Authorization: Bearer $TOKEN" -d "{}"

# L4-6/L4-7：断线续传 / 重启续传
#   记录 lastSequence → 断连/重启 → 重连 → 发 events.resume{afterSequence}

# L4-8：关闭配对 / 撤销设备踢出（≤1s 断言）
#   关闭配对：设置-设备 → 「关闭配对」→ 确认弹窗（= 撤销全部，令牌全失效）
#   撤销单设备：设备列表 → 撤销
#   WS 侧断言 error 帧（token-revoked）+ 连接关闭；重连 → 401

# L4-9：端口探测兜底
netstat -ano | findstr ":8765"          # 确认占用
# （起任意进程绑 8765 后）开启多端服务 → curl -k https://127.0.0.1:8766/v1/info

# 红线 1：无监听断言（服务关闭后）
netstat -ano | findstr "876[5-9]"

# 红线 4：日志扫描（手工）
#   %APPDATA%\Crish\AgentWorkbench\logs\ 下扫描：43 字符令牌形态 / CXN- / sk- 前缀
#   自动化：cargo test -p conex-telemetry（redact 库 + redact_log_scan 已入库，直接可跑）

# 红线 2/3/5/7：远程安全面（自动化）
cargo test -p agent-tools                 # 红线远程变体矩阵（G-9 已补）
cargo test -p crish-agent-workbench --test server_integration   # G-7 集成 11/11（含 ≤1s 撤销）

# 自动化门禁（验收前置，全绿基线）
cargo test --workspace
cargo test -p crish-agent-workbench        # server 模块 100+ 测试
npm run typecheck                          # 前端门禁
```
