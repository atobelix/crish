```rust
pub(crate) fn with_clock(pending, events, store, clock: Arc<AtomicU64>) -> Self {
    let shared = Arc::clone(&clock);
    Self { /* ... */, now: Arc::new(move || shared.load(Ordering::Relaxed)) }
}
```
- 时间基准替换点（全部走 `(self.now)()`）：`register` :348（expires_at_ms 计算）、`cancel` :406（cancelled_at_ms）、`on_event` :519、`restore` :439（到期判定）与 :467（remaining_ms）、`fire_timer` :653（fired_at_ms）。

**3.3.2 到期任务改 10ms 轮询（:600-608 `spawn_fire_task`）**：
```rust
fn spawn_fire_task(&self, declaration: TimerDeclaration) -> tokio::task::JoinHandle<()> {
    // ...组件 Arc 克隆同前，另克隆 now 与 expires_at_ms...
    tokio::spawn(async move {
        // 10ms 粒度轮询：now() >= expires_at_ms 即 fire（统一生产/测试
        // 一条路径，消除「注入时钟与 Instant 真实时钟混用」）

        interval.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
        while (now)() < expires_at_ms { interval.tick().await; }
        let outcome = std::panic::catch_unwind(AssertUnwindSafe(|| {
            fire_timer(&timers, &early_exit, &pending, &events, &store, &now, &declaration);
        }));

}
```

```rust
    pub(crate) fn unregister(&self, session_id: &str);

impl SecurityAuditSink for RoutingAuditSink {
    fn emit(&self, event: &AgentEvent) {
        let AgentEvent::SecurityAudit { session_id, decision, .. } = event else { return; };
        // CountingAuditSink :94-105 同构语义：denied 决策同步计数
        if *decision == SecurityAuditDecision::Denied { self.stats.record_security_audit_denied(); }

}
```
