### 2.1 `apps/crish-desktop/src/api/sessionEventProjection.ts` 防腐拦截

#### 1. 拦截识别器：`isContextStateBogusUserMessage`

精准识别由引擎历史 Bug 产生的 context-state 伪用户消息：

```typescript
/**
 * 校验是否为引擎历史 Bug 产生的 context-state 伪用户消息。
 * 判定依据：metadata.kind === 'context-state'，或无 client_message_id 且正文匹配 /^当前已切换到模型/。
 */
export function isContextStateBogusUserMessage(
  msg: AgentMessage | null | undefined
): boolean {
  if (!msg) return false;
  if (asRec(msg.metadata)?.kind === 'context-state') return true;
  const raw = msg as unknown as Record<string, unknown>;
  const clientMsgId = msg.clientMessageId ?? raw.client_message_id;
  return (
    !clientMsgId &&
    /^当前已切换到模型/.test(projectAgentMessageContent(msg).text.trim())
  );
}
```

### 2.2 `apps/crish-desktop/src/api/session-persistence/sessionHydrationMessages.ts` 历史自愈

#### 1. 孤儿状态对清洗器：`purgeStaleContextStateBubbles`

精准匹配旧 Bug 遗留的状态对：

- **消息 `i`（用户端幽灵消息）**：`sender === 'user'`，正文匹配 `/^(?:\[系统环境提示：\]?\s*)?当前已切换到模型/` 或 `metadata.kind === 'context-state'`。
- **消息 `i + 1`（伴生废话确认）**：`sender === 'assistant'`，且正文匹配 `/^(?:明白了[。.]?|收到[。.]?|好的[。.]?)?\s*当前配置[：:]/` 或 `当前状态已切换` / `已了解当前配置` / `收到环境更新` 等模板确认，且无有效工具调用。

```typescript
function isStaleContextStateUserMessage(m: SessionMessage): boolean {
  if (m.sender !== 'user') return false;
  if (
    (m.metadata as Record<string, unknown> | undefined)?.kind ===
    'context-state'
  )
    return true;
  return /^(?:\[系统环境提示：\]?\s*)?当前已切换到模型/.test(
    (m.text || '').trim()
  );
}

function isStaleAssistantBoilerplate(m: SessionMessage): boolean {
  if (
    m.sender !== 'assistant' ||
    (Array.isArray(m.toolCalls) && m.toolCalls.length > 0)
  )
    return false;
  const t = (m.text || '').trim();
  if (t.length > 500) return false;
  return /^(?:明白了[。.]?|收到[。.]?|好的[。.]?)?\s*(?:当前配置[：:]|(?:当前配置|当前状态已切换|已了解当前配置|我已了解|收到环境更新|(?:我已经了解|我已了解)?(?:已切换到模型|当前已切换到模型)))/i.test(
    t
  );
}

/**
 * 历史脏数据自愈净化：识别并移除由旧 Bug 生成的孤儿状态对：
 * (a) 消息为 sender === 'user' 且匹配 /^当前已切换到模型/；
 * (b) 紧邻消息为 sender === 'assistant' 且匹配确认配置废话回复。
 */
export function purgeStaleContextStateBubbles(
  messages: SessionMessage[]
): SessionMessage[] {
  if (!messages?.length) return [];
  const out: SessionMessage[] = [];
  for (let i = 0; i < messages.length; i++) {
    const curr = messages[i]!;
    if (isStaleContextStateUserMessage(curr)) {
      const next = messages[i + 1];
      if (next && isStaleAssistantBoilerplate(next)) i++; // 伴生废话确认回复一并净化
      continue; // 跳过伪用户消息自身
    }
    out.push(curr);
  }
  return out;
}
```
