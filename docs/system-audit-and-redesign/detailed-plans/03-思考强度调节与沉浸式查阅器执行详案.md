## 一、方案总纲与架构愿景

### 1.1 核心使命与战役定位

在以 **DeepSeek-R1**、**Claude 3.7 Sonnet (Extended Thinking)**、**OpenAI o1/o3-mini** 及 **Gemini 2.0 Flash Thinking** 为代表的现代深度推理大模型（Reasoning Models）开发与应用中，大模型已具备自主规划、深度反思、自我修正与长程工具编排的能力。

本详案（DIM-03）作为 Conex / Crish 全系统综合重构战役的关键一环，承担着以下三项决定性核心使命：

1. **打通思考强度全链路控制**：建立“全局模型默认（设置页）+ 会话即时覆盖（对话底栏）”双层控制，将用户指定的思考档位与 Token 预算精确下发至各厂商协议；
2. **重塑万字思考双轨查阅体验**：彻底废除 `4000` 字符硬截断，以“行内自适应弹性视口（8k~12k 预算，追尾防假死）+ 全量沉浸式查阅器（`FullThoughtModal`）”双轨架构，结合 **Worker 离屏计算防掉帧（RUL-05 裁决）**，呈现富文本 Markdown、TOC 目录跳转、全文检索与无损导出；
3. **根治 PromptCache 虚假“未命中”与断点缺失**：重构 `CacheMetricsBadge` 为四态语义（⚡ 缓存命中 / 💾 写入缓存 / 未命中 / 隐藏），在 Rust 端补齐 Anthropic `cache_control` 动态断点与 Beta Headers，确保长程会话享受 80%~95% 缓存命中加速与成本削减。

### 3.1 模块 1：PromptCache 协议补齐与四态徽标重构

#### 1. `crates/agent-provider/src/request/build_anthropic.rs`

- **文件路径**：`crates/agent-provider/src/request/build_anthropic.rs`
- **改动职责**：
  1. 在 `AnthropicRequestOptions` 中增加 `pub thinking_budget: Option<u64>`；
  2. 改造 `anthropic_thinking_object`：优先消费显式 `thinking_budget`，支持精准 Token 预算下发；
  3. 改造 `build_anthropic_request_body`：自动计算 `max_tokens >= budget_tokens + 4096` 安全抬升；
  4. 为 `tools` 列表的最后一个元素注入 `cache_control: { "type": "ephemeral" }` 断点；
  5. 校验 `system` 提示词末尾注入 `cache_control`。

```rust
// crates/agent-provider/src/request/build_anthropic.rs 代码规范

use super::cache_policy::allocate_breakpoints;

#[derive(Debug, Clone, Default)]
pub struct AnthropicRequestOptions {
    pub model: Option<String>,
    pub stream: Option<bool>,
    pub temperature: Option<f64>,
    pub max_tokens: Option<u64>,
    pub reasoning_effort: Option<ReasoningEffort>,
    pub thinking_intensity: Option<ThinkingIntensity>,
    /// 显式下发的精确 Token 预算（1024 ~ 64000）
    pub thinking_budget: Option<u64>,
    pub tool_choice: Option<ToolChoice>,
}

pub fn build_anthropic_request_body(
    opts: &AnthropicRequestOptions,
    context: &RequestBuildContext,
) -> Value {
    let mut body = Map::new();

    if let Some(model) = &opts.model {
        body.insert("model".to_string(), json!(model));
    }

    // 1. system prompt 构造与缓存断点注入
    let has_system = context.system_prompt.as_deref().map(str::trim).filter(|s| !s.is_empty()).is_some();
    let bp_plan = allocate_breakpoints(context.tools.len(), has_system, false);

    if let Some(prompt) = build_system_prompt_string(context.system_prompt.as_deref()) {
        if bp_plan.system {
            body.insert(
                "system".to_string(),
                json!([
                    {
                        "type": "text",
                        "text": prompt,
                        "cache_control": { "type": "ephemeral" }
                    }
                ]),
            );
        } else {
            body.insert("system".to_string(), json!(prompt));
        }
    }

    if let Some(stream) = opts.stream {
        body.insert("stream".to_string(), json!(stream));
    }
    if let Some(temperature) = opts.temperature {
        body.insert("temperature".to_string(), json!(temperature));
    }

    // 2. thinking 参数与 max_tokens 自动安全抬升 (max_tokens >= budget + 4096)
    let (thinking_obj, resolved_budget) = anthropic_thinking_resolved(opts);
    if let Some(thinking) = thinking_obj {
        body.insert("thinking".to_string(), thinking);
    }

    let min_max_tokens = resolved_budget.map(|b| b.saturating_add(4096)).unwrap_or(2048);
    let final_max_tokens = opts.max_tokens.unwrap_or(min_max_tokens).max(min_max_tokens);
    body.insert("max_tokens".to_string(), json!(final_max_tokens));

        body.insert(

            json!(context
                .tools
                .iter()
                .enumerate()
                .map(|(idx, tool)| {
                    let mut map = serde_json::Map::new();
                    map.insert("name".to_string(), json!(tool.name));
                    map.insert("description".to_string(), json!(tool.description));
                    map.insert("input_schema".to_string(), canonicalize_schema(&tool.effective_input_schema()));
                    if bp_plan.tools && (idx + 1 == tool_count) {
                        map.insert("cache_control".to_string(), json!({ "type": "ephemeral" }));
                    }
                    Value::Object(map)
                })
                .collect::<Vec<_>>()),
        );
        body.insert(
            "tool_choice".to_string(),
            anthropic_tool_choice(opts.tool_choice.as_ref().unwrap_or(&ToolChoice::Auto)),
        );
    }

    // 4. messages
    body.insert(
        "messages".to_string(),
        json!(to_anthropic_messages(context)),
    );

    Value::Object(body)
}

fn anthropic_thinking_resolved(opts: &AnthropicRequestOptions) -> (Option<Value>, Option<u64>) {
    let budget = if let Some(explicit_budget) = opts.thinking_budget {
        if explicit_budget == 0 {
            return (None, None);
        }
        explicit_budget
    } else if let Some(intensity) = opts.thinking_intensity {
        if intensity == ThinkingIntensity::Off {
            return (None, None);
        }
        thinking_to_budget_tokens(intensity)
    } else if let Some(effort) = opts.reasoning_effort {
        if matches!(effort, ReasoningEffort::Off | ReasoningEffort::Auto) {
            return (None, None);
        }
        effort_to_budget_tokens(effort)
    } else {
        return (None, None);
    };

    (
        Some(json!({
            "type": "enabled",
            "budget_tokens": budget,
        })),
        Some(budget),
    )
```

```ts
 * 消息列表窗口·DOM 预算与思考内容自适应切片引擎
 * 纯常量 + 纯函数，叶子模块。
 */

/** 行内常规展示的思考字符上限（从 4000 提升至 8000，保障段落完整） */
export const THINKING_DOM_CHAR_CAP = 8000;
/** 行内自适应允许的最大弹性字符上限 */
export const THINKING_DOM_CHAR_MAX = 12000;
/** 工具展开体写入 DOM 的字符上限 */
export const TOOL_OUTPUT_DOM_CHAR_CAP = 8000;
/** 工具展开体像素高度上限 */
export const TOOL_OUTPUT_PIXEL_CAP = 600;

export interface ThinkingBudgetResult {
  /** 用于行内渲染的安全文本（无污染尾缀） */
  displayText: string;
  /** 是否已被截断 */
  isTruncated: boolean;
  /** 完整原始字符数 */
  totalChars: number;
  /** 完整原始行数 */
  totalLines: number;
  /** 估算阅读时间（分钟） */
  readingMinutes: number;
}

/**
 * 思考内容智能切片算法：
 * 1. 文本在 cap 范围内，完整返回；
 * 2. 超 cap 时，在靠近 cap 处的换行符处优雅截断，避免断句撕裂；
 * 3. 返回结构化元数据供 UI 渲染渐变淡出遮罩与操作条。
 */
export function budgetThinkingText(
  text: string | undefined | null,
  cap: number = THINKING_DOM_CHAR_CAP
): ThinkingBudgetResult {
  const raw = text || '';
  const totalChars = raw.length;
  const lines = raw ? raw.split(/\r?\n/) : [];
  const totalLines = lines.length;
  const readingMinutes = Math.max(1, Math.round(totalChars / 500));

  if (!Number.isFinite(cap) || cap <= 0 || totalChars <= cap) {
    return {
      displayText: raw,
      isTruncated: false,
      totalChars,
      totalLines,
      readingMinutes,
    };
  }

  const safeCap = Math.min(Math.floor(cap), THINKING_DOM_CHAR_MAX);
  // 尝试在 safeCap 前 200 字符内寻找最近的换行符
  const sliceCandidate = raw.slice(0, safeCap);
  const lastNewline = sliceCandidate.lastIndexOf('\n');
  const cutIndex = lastNewline > safeCap - 200 ? lastNewline : safeCap;

  return {
    displayText: raw.slice(0, cutIndex),
    isTruncated: true,
    totalChars,
    totalLines,
    readingMinutes,
  };
}

/** 兼容旧代码签名的包装函数 */
export function budgetDomText(
  text: string,
  cap: number
): { text: string; truncated: boolean; originalLength: number } {
  const result = budgetThinkingText(text, cap);
  return {
    text: result.displayText,
    truncated: result.isTruncated,
    originalLength: result.totalChars,
  };
}
```

### 3.4 模块 4：RUL-05 Worker 离屏计算与 `FullThoughtModal` 沉浸式查阅器

#### 1. `apps/crish-desktop/src/workers/thoughtWorker.ts`

- **文件路径**：`apps/crish-desktop/src/workers/thoughtWorker.ts`
- **改动职责**：【**RUL-05 裁决核心落地**】将万字长文本的 Markdown 大纲（TOC）正则提取与全文关键词搜索移入 Web Worker 离屏线程，主线程 60fps 零卡顿。

```ts
/**
 * 万字长程思考 Worker 离屏计算引擎 (RUL-05 落地)
 * 处理 TOC 大纲提取、全文正则检索与关键词位置计算，避免阻塞 UI 主线程。
 */

export interface ThoughtHeading {
  id: string;
  level: number;
  text: string;
  lineIndex: number;
}

export interface ThoughtSearchResult {
  index: number;
  lineIndex: number;
  startChar: number;
  endChar: number;
  snippet: string;
}

export type ThoughtWorkerRequest =
  | { type: 'EXTRACT_TOC'; payload: { text: string } }
  | {
      type: 'SEARCH_TEXT';
      payload: { text: string; query: string; caseSensitive?: boolean };
    };

export type ThoughtWorkerResponse =
  | { type: 'TOC_RESULT'; payload: { headings: ThoughtHeading[] } }
  | { type: 'SEARCH_RESULT'; payload: { results: ThoughtSearchResult[] } };

self.onmessage = (e: MessageEvent<ThoughtWorkerRequest>) => {
  const req = e.data;
  if (req.type === 'EXTRACT_TOC') {
    const lines = req.payload.text.split(/\r?\n/);
    const headings: ThoughtHeading[] = [];

    for (let i = 0; i < lines.length; i++) {
      const line = lines[i];
      const match = line.match(/^(#{1,4})\s+(.+)$/);
      if (match) {
        const level = match[1].length;
        const text = match[2].trim();
        const id = `thought-heading-${i}-${text.slice(0, 16).replace(/\s+/g, '-')}`;
        headings.push({ id, level, text, lineIndex: i });
      }
    }
    self.postMessage({
      type: 'TOC_RESULT',
      payload: { headings },
    } satisfies ThoughtWorkerResponse);
  } else if (req.type === 'SEARCH_TEXT') {
    const { text, query, caseSensitive } = req.payload;
    if (!query || query.trim() === '') {
      self.postMessage({
        type: 'SEARCH_RESULT',
        payload: { results: [] },
      } satisfies ThoughtWorkerResponse);
      return;
    }

    const flags = caseSensitive ? 'g' : 'gi';
    const regex = new RegExp(
      query.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'),
      flags
    );
    const results: ThoughtSearchResult[] = [];
    const lines = text.split(/\r?\n/);

    let match: RegExpExecArray | null;
    let matchIdx = 0;

    // 按行高效检索
    let charOffset = 0;
    for (let lineIdx = 0; lineIdx < lines.length; lineIdx++) {
      const line = lines[lineIdx];
      while ((match = regex.exec(line)) !== null) {
        const start = match.index;
        const end = start + match[0].length;
        const snippetStart = Math.max(0, start - 20);
        const snippetEnd = Math.min(line.length, end + 20);
        const snippet =
          (snippetStart > 0 ? '...' : '') +
          line.slice(snippetStart, snippetEnd) +
          (snippetEnd < line.length ? '...' : '');

        results.push({
          index: matchIdx++,
          lineIndex: lineIdx,
          startChar: charOffset + start,
          endChar: charOffset + end,
          snippet,
        });
      }
      charOffset += line.length + 1;
    }

    self.postMessage({
      type: 'SEARCH_RESULT',
      payload: { results },
    } satisfies ThoughtWorkerResponse);
  }
};
```
