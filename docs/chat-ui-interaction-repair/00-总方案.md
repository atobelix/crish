## 二、模块一：输入框（Composer）防抖与多行切换体系

### 1. 核心数学模型与不变量

- **虚拟单行几何预算（Decoupled Baseline）**：
  $$\text{Capacity}_{\text{single}} = \max\Big(\text{MIN\_CAPACITY\_PX}, W_{\text{layout}} - W_{\text{controls\_single}} - \text{GUTTER}\Big)$$
  其中 $W_{\text{controls\_single}} = W_{\text{left}} + W_{\text{perm}} + W_{\text{right}}$ 为单行常态下控制按钮的固定宽度总和。判定容量时严禁使用多行展开后的通栏宽度。

- **双稳态施密特迟滞门限（Schmitt Trigger Deadband）**：
  - 展开上门限：$\text{Threshold}_{\text{expand}} = \text{Capacity}_{\text{single}}$（文本宽度 $\ge \text{Capacity}_{\text{single}}$ 时切多行）；
  - 收起下门限：$\text{Threshold}_{\text{collapse}} = \text{Capacity}_{\text{single}} - 24\text{px}$（文本宽度 $< \text{Capacity}_{\text{single}} - 24\text{px}$ 时切单行）；
  - 迟滞死区 $[\text{Capacity} - 24\text{px}, \text{Capacity}]$：保持上一帧状态，绝不翻转。

- **单例 Canvas 2D 离屏度量**：
  摒弃向 `document.body` 挂载 `<span>` 的高开销 DOM 操作，采用静态单例 Canvas 2D `ctx.measureText(cleanText).width` 进行 $O(1)$ 微秒级精确度量。

- **120ms 振荡自愈熔断器（Oscillation Breaker）**：
  若在 $120\text{ms}$ 时间窗口内检测到状态翻转次数 $\ge 3$ 次，立即触发熔断保护，强制锁定多行态 $400\text{ms}$，保护主渲染线程。

### 2. 代码实现规范（`useComposerMultiline.ts`）

```typescript
// apps/crish-desktop/src/components/chat-composer/useComposerMultiline.ts
import { useCallback, useLayoutEffect, useRef, useState } from 'react';

const GRID_GUTTER_PX = 18;
const MIN_CAPACITY_PX = 200;
const HYSTERESIS_BUFFER_PX = 24;
const FLIP_WINDOW_MS = 120;
const FLIP_THRESHOLD = 3;
const LOCKOUT_DURATION_MS = 400;

let sharedCanvas: HTMLCanvasElement | null = null;
let sharedCtx: CanvasRenderingContext2D | null = null;

function measureTextWidthCanvas(text: string, font: string): number {
  if (typeof window === 'undefined') return 0;
  if (!sharedCanvas) {
    sharedCanvas = document.createElement('canvas');
    sharedCtx = sharedCanvas.getContext('2d');
  }
  if (!sharedCtx) return text.length * 8;
  sharedCtx.font = font;
  return sharedCtx.measureText(text).width;
}

export function useComposerMultiline(opts: {
  text: string;
  hasAttachments: boolean;
  layoutRef: React.RefObject<HTMLElement | null>;
}): boolean {
  const { text, hasAttachments, layoutRef } = opts;
  const [multiline, setMultiline] = useState(() => hasAttachments || text.includes('\n'));
  const fontRef = useRef<string>('');
  const multilineRef = useRef(multiline);
  multilineRef.current = multiline;

  // 振荡熔断状态
  const flipHistoryRef = useRef<number[]>([]);
  const lockedUntilRef = useRef<number>(0);

  const measure = useCallback(() => {
    if (hasAttachments || text.includes('\n')) {
      setMultiline(true);
      return;
    }
    const layout = layoutRef.current;
    if (!layout) {
      setMultiline(false);
      return;
    }

    const now = performance.now();
    if (now < lockedUntilRef.current) {
      return; // 熔断锁定期间，保持多行
    }

    const left = layout.querySelector('[data-composer-grid="left"]') as HTMLElement | null;
    const perm = layout.querySelector('[data-composer-grid="perm"]') as HTMLElement | null;
    const right = layout.querySelector('[data-composer-grid="right"]') as HTMLElement | null;
    const editable = layout.querySelector('[contenteditable="true"], textarea') as HTMLElement | null;

    const layoutW = layout.clientWidth;
    const used =
      (left?.offsetWidth ?? 0) +
      (perm?.offsetWidth ?? 0) +
      (right?.offsetWidth ?? 0) +
      GRID_GUTTER_PX;
    const capacity = Math.max(MIN_CAPACITY_PX, layoutW - used);

    if (editable) {
      const cs = getComputedStyle(editable);
      fontRef.current = `${cs.fontWeight} ${cs.fontSize} ${cs.fontFamily}`;
    }

    const clean = text.replace(/\u200B/g, '').trim();
    if (!clean) {
      setMultiline(false);
      return;
    }

    const textWidth = measureTextWidthCanvas(clean, fontRef.current || '14px sans-serif');
    const isCurrentlyMulti = multilineRef.current;

    let nextState = isCurrentlyMulti;
    if (!isCurrentlyMulti && textWidth >= capacity) {
      nextState = true; // 突破上门限，展开
    } else if (isCurrentlyMulti && textWidth < capacity - HYSTERESIS_BUFFER_PX) {
      nextState = false; // 跌破下门限，收起
    }

    if (nextState !== isCurrentlyMulti) {
      // 记录翻转历史
      const history = flipHistoryRef.current.filter((t) => now - t < FLIP_WINDOW_MS);
      history.push(now);
      flipHistoryRef.current = history;

      if (history.length >= FLIP_THRESHOLD) {
        // 触发熔断
        lockedUntilRef.current = now + LOCKOUT_DURATION_MS;
        setMultiline(true);
        return;
      }
      setMultiline(nextState);
    }
  }, [hasAttachments, layoutRef, text]);

  useLayoutEffect(() => {
    measure();
    const el = layoutRef.current;
    if (!el || typeof ResizeObserver === 'undefined') return;
    const ro = new ResizeObserver(() => {
      window.requestAnimationFrame(() => measure());
    });
    ro.observe(el);
    return () => ro.disconnect();
  }, [layoutRef, measure]);

  return multiline;
}
```

## 三、模块二：FlowScrubber 滚动刻度体系重构

### 1. 架构解耦：Slot-Marker 模式

- **外层透明槽（`.tickSlot`）**：承担 Flex 垂直占比分配（`style={{ flexGrow: tick.flex }}`），背景完全透明，尺寸随段落高度伸缩；
- **内层刻度针（`.tick`）**：严格锁定 `height: 2px; flex: 0 0 2px; border-radius: 1px;`，纯水平细针，杜绝垂直拉伸；
- **隐形透明命中区（`::before` Hitbox）**：上下延伸 `8px`，向左延伸 `20px`，右侧 `right: 0` 严格禁止侵入右侧 14px 滚动条槽位。

### 2. 动效阶梯规范（100% 对齐 reference）

- 常态未激活：`width: 8px; opacity: 0.35;`
- 当前视口激活（`.isActive`）：`width: 16px; opacity: 0.6;`
- 轨道悬停未激活（`.root:hover .tick`）：`width: 16px; opacity: 0.5;`
- 轨道悬停激活项（`.root:hover .tick.isActive`）：`width: 24px; opacity: 0.8;`
- 刻度自身精准悬停（`.tick:hover`）：`width: 32px !important; opacity: 1;`
- 轨道悬停间距弹性展开：`.root:hover { gap: 12px; }`
