## 一、事故现场与因果断裂根因深度穿透

### 1.1 现象重现与表象病灶

在客户端（Desktop）冷重启或重新激活历史会话后，出现严重的对话倒置现象：

- **表象**：历史会话中所有的 Assistant 回复（AI 消息）全部聚集并排布在对话流最上方，而所有的 User 提问（用户消息）全部排布在对话流最下方。
- **对话交替逻辑断裂**：原本 `U1 -> A1 -> U2 -> A2 -> ...` 的交替式因果时序，在 UI 上彻底退化为 `[A1, A2, ...] -> [U1, U2, ...]`。

### 1.2 证据链排查与底层真相

通过对数据持久层、冷启动流程、切会话触发机制、投影解析面以及排序对比器的全链路穿透排查，确认了以下五环因果链：

#### 环 4：`compareMessageOrder` 兜底字典序导致的天然角色分流

- 在 [`messageOrderKey.ts:42-51`](file:///c:/Software/conex/apps/crish-desktop/src/store/messageOrderKey.ts#L42-L51) 中：
  ```ts

    a: MessageOrderKey,
    b: MessageOrderKey
  ): number {
    if (a.seq !== b.seq) return a.seq < b.seq ? -1 : 1;
    if (a.ts !== b.ts) return a.ts < b.ts ? -1 : 1;
    return a.id.localeCompare(b.id);
  }
  ```
- 当所有消息的 `seq` 均为 `MAX_SAFE_INTEGER`（投影未保留事件 sequence），且 `ts` 均为相同的 `Date.now()` 时，排序逻辑**退化到第三重比较：`a.id.localeCompare(b.id)`**。
- 系统中的消息 ID 规则：
  - 助手消息 ID：`assistant-msg-*` 或 `hyd-asst-*`
  - 用户消息 ID：`user-msg-*` 或 `hyd-user-*`
- 在 ASCII / Unicode 编码中，`'a' < 'u'` 且 `'hyd-a' < 'hyd-u'`！
- **灾难性后果**：所有 Assistant 消息由于首字母 'a'，在字典序上恒小于所有 User 消息的 'u'。二分插入和全局排序使得所有 Assistant 消息被强制排在前端，User 消息被排在尾部。

## 二、架构重构目标与反退避原则

根据 `AGENTS.md` 与 `.agents/LOGIC_PLAN_NO_RETREAT.md` 规范：

1. **零临时补丁原则**：严禁在 UI 渲染层通过 `if (sender === 'user')` 进行无序魔术补偿，必须穿透到底层数据水合与排序键架构。
2. **单一真理源（SSOT）单轨**：本地 SQLite 数据库作为历史持久化的唯一真源；切会话必须优先水合 SQLite，仅在本地完全为空时作为保底拉取引擎事件。
3. **因果确定性不可逆**：无论网络状况、事件重放还是时间戳抖动，同一轮对话中用户提问必然先于助手回答（Causal Precedence: User $\prec$ Assistant）。

## 三、四大核心模块重构详案

### 3.1 模块一：`sessionEventProjection.ts` 时间戳权威提取与序号透传

#### 3.1.1 根治策略

1. 废除无条件 fallback `Date.now()` 的盲盒逻辑。
2. 建立精准的多层时间戳解析器 `extractEventTimestamp`：
   - 第一优先级：消息载荷自带的真实创建时间 `agent.createdAt`（标准 ISO8601 字符串）或 `agent.timestamp`；
   - 第二优先级：事件内层元数据 `inner.timestamp` 或 `inner.createdAt`；
   - 第三优先级：外层信封元数据 `ev.timestamp`；
   - 兜底回退：若且仅若所有元数据均缺失时，标记为未定义或递增相对时间，坚决禁止使用单一瞬时 `Date.now()` 污染全量历史。
3. 在投影生成 `SessionMessage` 时，完整保留并绑定引擎事件的 `sequence` 到 `engineSequence` 字段，使排序器获得物理事件时序保证。

#### 3.1.2 落地改动实现设计

```typescript
// c:\Software\conex\apps\crish-desktop\src\api\sessionEventProjection.ts

/**
 * 从消息载荷与事件信封中提取精准历史时间戳。
 * 遵循严格优先级：msg.createdAt > msg.timestamp > inner.timestamp > ev.timestamp。
 * 若均无有效时间，返回 undefined 由调用方根据因果链安全处理，避免 Date.now() 冲刷历史。
 */
export function extractMessageTimestamp(
  msg: AgentMessage | null | undefined,
  inner: AnyRec | null,
  outerEv?: AnyRec | null
): number | undefined {
  // 1. 优先读取 AgentMessage 载荷中的权威创建时间
  if (msg) {
    const rawCreated = msg.createdAt ?? (msg as unknown as AnyRec).created_at;
    if (typeof rawCreated === 'string') {
      const parsed = Date.parse(rawCreated);
      if (!Number.isNaN(parsed) && parsed > 0) return parsed;
    }
    const rawMsgTs = (msg as unknown as AnyRec).timestamp;
    if (
      typeof rawMsgTs === 'number' &&
      Number.isFinite(rawMsgTs) &&
      rawMsgTs > 0
    ) {
      return rawMsgTs;
    }
  }

  // 2. 次优读取事件内层元数据
  if (inner) {
    const rawTs = inner.timestamp;
    if (typeof rawTs === 'string') {
      const parsed = Date.parse(rawTs);
      if (!Number.isNaN(parsed) && parsed > 0) return parsed;
    }
    if (typeof rawTs === 'number' && Number.isFinite(rawTs) && rawTs > 0) {
      return rawTs;
    }
  }

  // 3. 再次读取外层信封
  if (outerEv) {
    const rawOuterTs = outerEv.timestamp;
    if (
      typeof rawOuterTs === 'number' &&
      Number.isFinite(rawOuterTs) &&
      rawOuterTs > 0
    ) {
      return rawOuterTs;
    }
  }

  return undefined;
}
```

### 3.2 模块二：重构会话懒水合链路（SQLite SSOT 第一真源）

#### 3.2.1 链路重构原则

- **明确真理源梯次**：
  $$\text{本地已水合内存态} \xrightarrow{\text{未就绪}} \text{本地 SQLite (ensureSessionBodyHydrated)} \xrightarrow{\text{记录为空}} \text{远程引擎 (hydrateSessionMessagesFromEngine)}$$
- **竞态防护（Generation Token）**：
  引入 `bumpHydrateGeneration` 与 `isHydrateGenerationCurrent`，在快速连续切换会话（A $\rightarrow$ B $\rightarrow$ C）时，丢弃所有已过期的异步 IO 回调，杜绝陈旧数据回流到当前活跃会话。
- **状态原子收敛**：
  SQLite 加载完成后，立即置位 `messagesHydrated = true`。空会话亦标为已水合，避免重复触发。

```typescript
import { isPendingDraft } from './draftSession';
import type { SessionImmerSet, SessionStoreGet } from './sessionActions';
import { getSessionById, getSessionForWrite } from './sessionIndex';
import { maybeAutoTitleSession } from './sessionSharedHelpers';
import {
  bumpHydrateGeneration,
  isHydrateGenerationCurrent,
  ensureSessionBodyHydrated,
} from '../api/session-persistence/sessionRustDbHydrate';
import { logger, toErrMeta } from '../services/LoggerService';

/**
 * 跨模块可复用的会话正文懒水合触发器（SSOT 单轨版）。
 *
 * 核心调度流水线：
 * 1. 检查会话状态：已完整水合且有正文则快速跳过（O(1)）；
 * 2. 签发单调递增 Generation Token，拦截快速切会话竞态；
 * 3. 【第一真源】：调用 SQLite 的 ensureSessionBodyHydrated(sessionId)；
 * 4. 【回退通道】：仅在 SQLite 返回空消息时，回退调用引擎事件 hydrateSessionMessagesFromEngine；
 * 5. 校验 Token 时效性，原子提交至 Immer Store。

export async function triggerSessionMessageHydrateIfNeeded(
  sessionId: string,
  get: SessionStoreGet,
  set: SessionImmerSet
): Promise<void> {
  if (!sessionId || isPendingDraft(sessionId)) return;

  const state = get();
  const sess = getSessionById(state, sessionId);
  if (!sess) return;

  // 卫语句：已经水合过且正文非空，无需重复触发
  if (sess.messagesHydrated && sess.messages && sess.messages.length > 0) {
    return;
  }

  // 签发最新世代令牌，防连续切换竞态
  const genToken = bumpHydrateGeneration();

  try {
    // ── 步骤一：优先从本地 SQLite 数据库（SSOT）加载 ──
    const loadedFromDb = await ensureSessionBodyHydrated(sessionId);

    // 检查响应是否仍处于最新世代（若用户已切换到其他会话，直接丢弃）
    if (!isHydrateGenerationCurrent(genToken)) {
      return;
    }

    if (loadedFromDb.messages && loadedFromDb.messages.length > 0) {
      set((draft) => {
        const s = getSessionForWrite(draft, sessionId);
        if (!s) return;
        s.messages = loadedFromDb.messages;
        s.agents = loadedFromDb.agents;
        s.messagesHydrated = true;
        // 本地落库数据不依赖 engineCursor 覆盖，保持既有或由后续增量接管
        maybeAutoTitleSession(draft, s);
        draft.sessionsById[sessionId] = s;
      });
      return;
    }

    // ── 步骤二：本地 SQLite 无数据，回退从引擎 session-events 投影 ──
    const { hydrateSessionMessagesFromEngine, mergeMessagesById } =
      await import('../api/sessionEventHydrate');

    const res = await hydrateSessionMessagesFromEngine(sessionId);

    if (!isHydrateGenerationCurrent(genToken)) {
      return;
    }

    if (res.ok) {
      set((draft) => {
        const s = getSessionForWrite(draft, sessionId);
        if (!s) return;
        if (res.messages.length > 0) {
          s.messages = mergeMessagesById(s.messages || [], res.messages);
          maybeAutoTitleSession(draft, s);
        }
        s.engineCursor = Math.max(s.engineCursor ?? 0, res.nextAfterSequence);
        if (res.lastRunId) s.engineLastRunId = res.lastRunId;
        s.hydratedFromEngine = true;
        s.messagesHydrated = true;
        if (res.truncated === true) {
          s.hydrateTruncated = true;
        } else {
          delete s.hydrateTruncated;
        }
        draft.sessionsById[sessionId] = s;
      });
    }
  } catch (err) {
    logger.error('[hydrate] triggerSessionMessageHydrate failed', {
      module: 'sessionActivateHydrate',
      sessionId,
      err: toErrMeta(err),
    });
  }
}
```

### 3.3 模块三：改进 `compareMessageOrder` 的 Tie-Break 规则

#### 3.3.1 字典序缺陷的彻底清退

原对比器仅有三级：
$$\text{seq (小在前)} \rightarrow \text{ts (小在前)} \rightarrow \text{id.localeCompare(b.id)}$$
在时间戳相同时，无条件依赖 `a.id.localeCompare(b.id)` 会由于 ID 前缀模式（`assistant-*` 与 `user-*`）导致角色天然被割裂。

#### 3.3.2 四级因果全序（Causal Total Order）模型

重构后的排序键必须引入**对话因果角色优先权（Causal Role Rank）** 与 **自然数值 ID 排序（Natural Numeric ID Order）**：

1. **第一优先级（物理事件序号 `seq`）**：
   - 只要双方均具有有效的 `engineSequence`，严格以事件生成序号为准；小在前。
2. **第二优先级（时间戳 `ts`）**：
   - 双方 `seq` 缺省（同为 `MAX_SAFE_INTEGER`）时，按物理时间戳升序排序；小在前。
3. **第三优先级（因果角色权 `roleRank`）**：
   - 当时间戳完全一致（处于同一毫秒或并发产生）时，根据对话交互的因果定律裁决：
     - System 消息先于 User 消息先于 Assistant 消息；
     - 权值分配：`system = 0`，`user = 1`，`assistant = 2`，`other = 3`；
     - **断言定理**：在同一时间戳下，User（提问）永远排在 Assistant（回答）之前，杜绝 AI 消息冒顶。
4. **第四优先级（自然数字 ID 排序 `naturalIdCompare`）**：
   - 仅当角色的权值亦相同时（如连续两条 Assistant 消息），进行 ID 终极平局决胜；
   - 采用 `{ numeric: true, sensitivity: 'base' }` 的自然数字排序，使 `hyd-user-2` 排在 `hyd-user-10` 之前，避免纯 ASCII 字典序颠倒序列号。

#### 3.3.3 `messageOrderKey.ts` 完整重写方案

```typescript
// c:\Software\conex\apps\crish-desktop\src\store\messageOrderKey.ts

/**
 * 消息时间线因果保序（Causal Order SSOT）。
 *
 * 唯一排序四级全序规范：
 * 1. seq（引擎物理序号 engineSequence）
 * 2. ts（精确创建时间戳 timestamp）
 * 3. roleRank（因果角色权重：system=0, user=1, assistant=2）
 * 4. id（自然数字 ID 排序：numeric=true）
 */

export type MessageRoleRank = 0 | 1 | 2 | 3;

export function resolveRoleRank(sender?: string | null): MessageRoleRank {
  if (!sender) return 3;
  switch (sender.toLowerCase()) {
    case 'system':
      return 0;
    case 'user':
      return 1;
    case 'assistant':
      return 2;
    default:
      return 3;
  }
}

/** 排序键定义 */
export type MessageOrderKey = {
  seq: number;
  ts: number;
  roleRank: MessageRoleRank;
  id: string;
};

/**
 * 从消息对象提取全序比较键。
 */
export function messageOrderKey(m: {
  timestamp?: number;
  id?: string;
  engineSequence?: number;
  sender?: string;
}): MessageOrderKey {
  const ts =
    typeof m.timestamp === 'number' &&
    Number.isFinite(m.timestamp) &&
    m.timestamp > 0
      ? m.timestamp
      : 0;

  const rawSeq = m.engineSequence;
  const seq =
    typeof rawSeq === 'number' && Number.isFinite(rawSeq)
      ? rawSeq
      : Number.MAX_SAFE_INTEGER;

  const roleRank = resolveRoleRank(m.sender);
  const id = typeof m.id === 'string' ? m.id : '';

  return { seq, ts, roleRank, id };
}

/**
 * 四级全序比较器：
 * seq 优先 → ts 次之 → roleRank 因果平局决胜 → id 自然数字序终极兜底。
 */
export function compareMessageOrder(
  a: MessageOrderKey,
  b: MessageOrderKey
): number {
  // 1. 物理序号比较
  if (a.seq !== b.seq) {
    return a.seq < b.seq ? -1 : 1;
  }

  // 2. 时间戳比较
  if (a.ts !== b.ts) {
    return a.ts < b.ts ? -1 : 1;
  }

  // 3. 关键修正：因果角色比较（User 恒先于 Assistant，杜绝 AI 消息冒顶）
  if (a.roleRank !== b.roleRank) {
    return a.roleRank < b.roleRank ? -1 : 1;
  }

  // 4. 自然数字 ID 比较（如 hyd-user-2 位于 hyd-user-10 之前）
  return a.id.localeCompare(b.id, undefined, {
    numeric: true,
    sensitivity: 'base',
  });
}

/** 列表是否已按 compareMessageOrder 非降序排列 */
export function isSortedByMessageOrder(
  list: readonly {
    timestamp?: number;
    id?: string;
    sender?: string;
    engineSequence?: number;
  }[]
): boolean {
  for (let i = 1; i < list.length; i++) {
    const prev = messageOrderKey(list[i - 1]!);
    const curr = messageOrderKey(list[i]!);
    if (compareMessageOrder(prev, curr) > 0) return false;
  }
  return true;
}

/** 二分插入点查找 */
export function findInsertIndex(
  list: readonly MessageOrderKey[],
  item: MessageOrderKey
): number {
  let lo = 0;
  let hi = list.length;
  while (lo < hi) {
    const mid = (lo + hi) >>> 1;
    if (compareMessageOrder(list[mid]!, item) <= 0) {
      lo = mid + 1;
    } else {
      hi = mid;
    }
  }
  return lo;
}
```
