format_markdown.js

JavaScript 技能

自动规范化 Markdown 格式，修复空行、列表对齐、标题间距，并优化中文和英文之间的空格间隔。

技能实现源码：
// format_markdown.js
export default function formatMarkdown(text) {
  // 替换中英文之间的空格
  let formatted = text.replace(/([\u4e00-\u9fa5])([A-Za-z0-9])/g, '$1 $2');
  formatted = formatted.replace(/([A-Za-z0-9])([\u4e00-\u9fa5])/g, '$1 $2');
  // 规范化空行
  formatted = formatted.replace(/\n{3,}/g, '\n\n');
  return formatted;
}
